cmake_minimum_required(VERSION 3.15)
project(ParseEngineLsp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(parseengine_lsp STATIC json.cpp grammar.cpp lsp.cpp)
target_include_directories(parseengine_lsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(parseengine_lsp PRIVATE /W4 /utf-8)
else()
  target_compile_options(parseengine_lsp PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(lsp-server main.cpp)
target_link_libraries(lsp-server PRIVATE parseengine_lsp)
if(MSVC)
  target_compile_options(lsp-server PRIVATE /utf-8)
endif()

enable_testing()
add_executable(lsp-tests test_lsp.cpp)
target_link_libraries(lsp-tests PRIVATE parseengine_lsp)
if(MSVC)
  target_compile_options(lsp-tests PRIVATE /utf-8)
endif()
add_test(NAME lsp-tests COMMAND lsp-tests)

find_package(Threads REQUIRED)
target_link_libraries(parseengine_lsp PUBLIC Threads::Threads)
