cmake_minimum_required(VERSION 3.15)
project(ParseEngineSimpleAst LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(parseengine_simple_ast STATIC simple_ast.cpp)
target_include_directories(parseengine_simple_ast PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(parseengine_simple_ast PRIVATE /W4 /utf-8)
else()
  target_compile_options(parseengine_simple_ast PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(simple_ast-cli main.cpp)
target_link_libraries(simple_ast-cli PRIVATE parseengine_simple_ast)
enable_testing()
add_executable(simple_ast-tests test_simple_ast.cpp)
target_link_libraries(simple_ast-tests PRIVATE parseengine_simple_ast)
add_test(NAME simple_ast-tests COMMAND simple_ast-tests)
