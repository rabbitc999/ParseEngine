cmake_minimum_required(VERSION 3.16)
project(ParseEngineAllocator LANGUAGES C CXX)
add_library(allocator INTERFACE)
add_library(ParseEngine::Allocator ALIAS allocator)
target_include_directories(allocator INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_features(allocator INTERFACE cxx_std_17)
add_executable(allocatorDemo main.cpp)
target_link_libraries(allocatorDemo PRIVATE allocator)
include(CTest)
if(BUILD_TESTING)
    foreach(mode Linked Linear)
        add_executable(pool${mode} testPool.c)
        set_target_properties(pool${mode} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
        add_executable(containers${mode} testAllocator.cpp)
        target_link_libraries(containers${mode} PRIVATE allocator)
        if(mode STREQUAL "Linear")
            target_compile_definitions(pool${mode} PRIVATE MEMORY_ALLOCATOR_TYPE=MEMORY_ALLOCATOR_LINEAR)
            target_compile_definitions(containers${mode} PRIVATE MEMORY_ALLOCATOR_TYPE=MEMORY_ALLOCATOR_LINEAR)
        endif()
        add_test(NAME pool${mode} COMMAND pool${mode})
        add_test(NAME containers${mode} COMMAND containers${mode})
    endforeach()
endif()
